# scripts/simulation.gd:735 skill_info
func skill_info(skill: String) -> Dictionary:
	var info: Dictionary=_progression.info(skill,_skills.get(skill,{}),_config,souls,spirit)
	if skill=="climate" and not info.is_empty():
		info.effect_values["temperature_mode"]=str(climate_options.temperature)
		info.effect_values["cloud_mode"]=str(climate_options.clouds)
	if skill=="ghost_parade" and not info.is_empty():
		info.cost=maxf(5.0,spirit)
		info.effect_values["spirit_budget"]=spirit
		info.effect_values["summon_count"]=floori(spirit/5.0)
	if str(_config.get("game_mode",""))=="tutorial" and not info.is_empty():
		info.unlocked=skill in _config.get("tutorial_allowed_skills",[])
		info.can_unlock=false
		info.can_upgrade=false
		info.upgrade_reason="引导中由神明授予技能，无需购买升级。" if info.unlocked else "这位神明尚未登场。"
	if not outcome.is_empty() and not info.is_empty():
		info.can_upgrade=false
		info.upgrade_reason="本局已经结束"
	return info


# scripts/simulation.gd:754 upgrade_skill
func upgrade_skill(skill: String) -> Dictionary:
	if bool(_config.pantheon_enabled) and skill not in PantheonCatalog.skills(): return {"ok":false,"reason":"该技能已并入三神九技能，不再单独升级。","cost":0.0}
	if str(_config.get("game_mode",""))=="tutorial": return {"ok":false,"reason":"引导中由神明授予技能，无需购买升级。","cost":0.0}
	if not outcome.is_empty(): return {"ok":false,"reason":"本局已经结束","cost":0.0}
	var result: Dictionary=_progression.upgrade(skill,_skills.get(skill,{}),_config,souls,spirit)
	if not bool(result.ok): return result
	spirit-=float(result.cost)
	spirit_sources.spent_upgrades=float(spirit_sources.spent_upgrades)+float(result.cost)
	_event(("与神明合作：解锁%s，消耗 %.0f 灵力。"%[_skills[skill].title,float(result.cost)]) if bool(result.get("unlocked_now",false)) else ("%s升至 %d 级，消耗 %.0f 灵力；累计亡魂不消耗。"%[_skills[skill].title,int(result.level),float(result.cost)]))
	return result

func _record_deaths(index: int, amount: float, spatial: bool = false, cause: String = "natural") -> void:
	var nation: Dictionary = nations[index]
	var available: int = int(floor(maxf(0.0, float(nation.population) - (0.0 if spatial else _protected_population(index)))))
	var actual: int = 0
	if spatial:
		actual = clampi(int(amount), 0, available)
	else:
		var key: String = "starvation_credit" if cause == "starvation" else "death_credit"
		nation[key] = float(nation[key]) + clampf(amount, 0.0, float(available))
		actual = mini(available, int(floor(float(nation[key]) + 0.00000001)))
		nation[key] = float(nation[key]) - float(actual)
	if actual <= 0: return
	nation.population = float(nation.population) - float(actual)
	nation.deaths = float(nation.deaths) + float(actual)
	if cause == "starvation": nation.starvation_deaths = float(nation.starvation_deaths) + float(actual)
	if cause in ["wildlife","dinosaur"]: nation.wildlife_deaths = int(nation.get("wildlife_deaths", 0)) + actual
	if cause == "dinosaur": nation.dinosaur_deaths = int(nation.get("dinosaur_deaths",0)) + actual
	if cause == "fire": nation.fire_deaths = int(nation.get("fire_deaths",0)) + actual
	var before_souls: int=souls
	souls += actual
	if bool(_config.progression_enabled) and not bool(_config.pantheon_enabled):
		for key: String in DivinityProgression.DEFINITIONS:
			var threshold: int=int(DivinityProgression.DEFINITIONS[key].unlock)
			if _skills.has(key) and before_souls<threshold and souls>=threshold: _event("累计收集 %d 魂，解锁%s；亡魂指标不消耗。"%[threshold,str(_skills[key].title)])
	var soul_income: float=float(actual) * float(_config.spirit_per_soul)
	spirit += soul_income
	spirit_sources.souls=float(spirit_sources.get("souls",0.0))+soul_income
	var actual_counts: Array=[0,0]
	actual_counts[index]=actual
	human_deaths_recorded.emit(cause,actual_counts)
	if not spatial:
		_faith_death_serial+=1
		note_faith_event(index,"death",minf(1.0,float(actual)/maxf(1.0,float(nation.population))*3.0),"death_%d"%_faith_death_serial)


# scripts/simulation.gd:3146 _check_outcome
func _check_outcome() -> void:
	if str(_config.get("game_mode",""))=="tutorial" and str(_config.get("tutorial_phase","")) in ["final_cast","finale","complete"]: return
	if not outcome.is_empty(): return
	if target_souls > 0 and souls >= target_souls:
		outcome = "won"
		_event("考核通过！累计收集 %s 亡魂。" % souls)
	elif float(nations[0].population) + float(nations[1].population) < 1.0:
		outcome = "extinct"
		_event("双岛人类灭绝。可重新开始培养新的聚落。")
	elif bool(_config.science_ending) and _technology_finished():
		outcome = "science"
		_event(("封魂设施实际建成，本次沙盒演化结束。" if bool(_config.civilization_enabled) else "文明科技达到 100%，本次沙盒演化结束。") if str(_config.get("game_mode", "")) == "sandbox" else "灵魂封存工程完成，人类亡魂不再进入上帝的收魂考核。")
	elif max_years > 0.0 and year >= max_years - 0.000001:
		year = max_years
		outcome = "time"
		_event("考核期限已到，收魂指标未完成。")


# scripts/simulation.gd:3189 set_tutorial_permissions
func set_tutorial_permissions(skills: Array, phase: String) -> void:
	if str(_config.get("game_mode",""))!="tutorial": return
	var allowed: Array[String]=[]
	for key: String in skills:
		if _skills.has(key) and not allowed.has(key): allowed.append(key)
	_config["tutorial_allowed_skills"]=allowed
	_config["tutorial_phase"]=phase
